async function onPaused({ requestId, request }) {
  const u = new URL(request.url);
  const p = u.pathname.replace(/^.*\/api\/v1/, "");
  const th = Object.entries(request.headers).find(([k]) => k.toLowerCase() === "x-tenant-id")?.[1] ?? null;
  const body = request.postData ? JSON.parse(request.postData) : null;
  captured.push({ method: request.method, path: p, query: u.search, tenant: th, body });
  if (request.method === "OPTIONS") return send("Fetch.fulfillRequest", { requestId, responseCode: 204, responseHeaders: cors(), body: "" });
  const ok = (data, total) => ({ code: 200, msg: "Success", data, ...(total !== undefined ? { total } : {}) });
  const err = (status, errorCode, msg) => [status, { code: status, msg, data: null, total: null, extend: { errorCode, requestId: `mock-${errorCode}` } }];
  let status = 200, res;
  const m = p.match(/^\/roles\/(\d+)\/permissions$/);
  if (p === "/auth/login") res = ok({ accessToken: jwt(), refreshToken: jwt(), tokenType: "Bearer", expiresIn: 3600 });
  else if (p === "/me") res = ok({ id: identity.userId, displayName: "Synthetic user", email: "synthetic@example.invalid" });
  else if (p === "/me/permissions") res = ok(identity);
  else if (p === "/platform/tenants") res = ok(TENANTS, 1);
  else if (p === "/platform/switch-tenant") res = ok({ tenantId: 1, code: "ACME", name: "Acme Logistics", header: "X-Tenant-Id" });
  else if (p === "/roles" && request.method === "GET") res = ok(ROLES.map((r) => (r.id === 2 ? { ...r, permissionCodes: rolePerms[2] } : r)), ROLES.length);
  else if (p === "/roles" && request.method === "POST") {
    if (failCreateRole) [status, res] = err(409, "ROLE_CODE_TAKEN", "Role code already exists (MOCKED)");
    else { status = 201; res = { code: 201, msg: "Created", data: { id: 77, code: body.code, description: body.description, protectedRole: false, permissionCodes: [] } }; }
  } else if (m && request.method === "GET") res = ok({ roleId: Number(m[1]), roleCode: "x", fullAccess: m[1] === "1", editable: m[1] !== "1", permissionCodes: rolePerms[m[1]] ?? [] });
  else if (m && request.method === "POST") { rolePerms[m[1]] = [...new Set([...(rolePerms[m[1]] ?? []), ...body.permissionCodes])]; res = ok({ roleId: Number(m[1]), permissionCodes: rolePerms[m[1]] }); }
  else if (m && request.method === "DELETE") {
    if (failDeletePerms) [status, res] = err(409, "OPERATION_NOT_ALLOWED", "Removal refused (MOCKED)");
    else { const codes = u.searchParams.getAll("codes"); rolePerms[m[1]] = (rolePerms[m[1]] ?? []).filter((c) => !codes.includes(c)); res = ok({ roleId: Number(m[1]), permissionCodes: rolePerms[m[1]] }); }
  } else if (p === "/permissions" && request.method === "GET") res = ok(CATALOG, CATALOG.length);
  else if (p.startsWith("/permissions/") && request.method === "DELETE") [status, res] = err(409, "RESOURCE_IN_USE", "Permission in use (MOCKED)");
  else res = ok(null);
  await send("Fetch.fulfillRequest", { requestId, responseCode: status, responseHeaders: [...cors(), { name: "Content-Type", value: "application/json" }], body: Buffer.from(JSON.stringify(res)).toString("base64") });
}

async function shot(name, full = false) {
  let extra = {};
