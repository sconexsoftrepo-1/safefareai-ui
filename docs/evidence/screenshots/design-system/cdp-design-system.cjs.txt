async function onPaused({ requestId, request }) {
  const p = new URL(request.url).pathname.replace(/^.*\/api\/v1/, "");
  if (request.method === "OPTIONS") return send("Fetch.fulfillRequest", { requestId, responseCode: 204, responseHeaders: cors(), body: "" });
  const ok = (data, total) => ({ code: 200, msg: "Success", data, ...(total !== undefined ? { total } : {}) });
  const list = (a) => ok(a, a.length);
  let res;
  const id = ACCOUNTS[current];
  if (p === "/me") res = ok({ id: id.userId, displayName: `Synthetic ${current}`, email: `${current}@example.invalid`, tenantName: "Ganesh Travels" });
  else if (p === "/me/permissions") res = ok(id);
  else if (p === "/platform/tenants") res = list(TENANTS);
  else if (p === "/groups") res = list([{ id: 1, name: "North depot" }, { id: 2, name: "South depot" }]);
  else if (p === "/dashboard/overview") res = ok(OVERVIEW);
  else if (p === "/dashboard/fleet-status") res = list(STATUS);
  else if (p === "/dashboard/trends") res = list(TRENDS);
  else if (/^\/dashboard\/safety/.test(p)) res = ok(SAFETY);
  else if (p === "/devices") res = list(DEVICES);
  else if (p === "/alarms" || p === "/alerts") res = list(ALERTS);
  else if (p === "/roles") res = list(ROLES.map((r) => ({ ...r, tenantId: null })));
  else if (/^\/roles\/\d+\/permissions$/.test(p)) res = ok({ roleId: Number(p.split("/")[2]), permissionCodes: ["device.read", "alarm.read"], editable: true, fullAccess: false });
  else if (/^\/roles\/4\/menu-permissions$/.test(p)) { const dto = JSON.parse(fs.readFileSync(`${ROOT}/tests/fixtures/driver-role-menu-permissions.json`, "utf8")); delete dto._source; res = ok(dto); }
  else if (p === "/platform/switch-tenant") { const body = JSON.parse(request.postData); const t = TENANTS.find((x) => x.id === body.tenantId); res = ok({ tenantId: t.id, code: t.code, name: t.name, header: "X-Tenant-Id" }); }
  else if (p === "/permissions") res = list(PERMISSIONS);
  else if (p === "/driver/me") res = ok({ id: 1, name: "Synthetic driver", email: "driver@example.invalid", phone: "+1 555 0100", status: "ACTIVE", licence: { status: "VALID", expiresOn: "2028-01-01" } });
  else if (p === "/driver/assignment") res = ok({ currentVehicle: { deviceId: 7, deviceName: "Coach 12" }, nextTrip: TRIPS[0], today: TRIPS });
  else if (p === "/driver/dispatches") res = list(TRIPS);
  else if (p === "/driver/shifts/current") res = ok(null);
  else if (p === "/driver/working-hours") res = ok({ summary: { totalHours: 38.5, shiftCount: 5, averageSeconds: 27720, longestSeconds: 32400 }, shifts: [] });
  else if (p === "/driver/leave-requests") res = list(LEAVE);
  else if (p === "/driver/checklist-template") res = list(TEMPLATE);
  else if (p === "/driver/safety-score") res = ok({ score: { score: 91, grade: "A", alarmCount: 3, severeCount: 0, penaltyPoints: 2, tripCount: 21, distanceKm: 1840 }, byType: [{ label: "Harsh braking", count: 2 }, { label: "Overspeed", count: 1 }] });
  else { unmatched.add(p); res = ok([], 0); }
  await send("Fetch.fulfillRequest", { requestId, responseCode: 200, responseHeaders: [...cors(), { name: "Content-Type", value: "application/json" }], body: Buffer.from(JSON.stringify(res)).toString("base64") });
}

async function viewport(w, h) { await send("Emulation.setDeviceMetricsOverride", { width: w, height: h, deviceScaleFactor: 1, mobile: w < 768 }); }
async function go(url, ms = 1900) { await send("Page.navigate", { url }); await sleep(ms); }
async function shot(name, full = false) {
  const r = await send("Page.captureScreenshot", { format: "png", captureBeyondViewport: full });
  fs.writeFileSync(path.join(OUT, `${name}.png`), Buffer.from(r.data, "base64"));
}
async function as(account) {
  current = account;
  await go(`${APP}/login`, 700);
  await evaluate(`sessionStorage.setItem('safefare:session:v1', ${JSON.stringify(JSON.stringify({ accessToken: jwt(), tokenType: "Bearer", accessExpiresAt: Date.now() + 3600_000, refreshToken: "synthetic", refreshExpiresAt: null }))})`);
}
const PROBE = `(() => {
  const families = {};
  for (const el of document.querySelectorAll('main *, nav *, header *')) { if (!el.childNodes.length || !el.textContent.trim()) continue; const f = getComputedStyle(el).fontFamily.split(',')[0].replace(/"/g, ''); families[f] = (families[f] ?? 0) + 1; }
  const h1 = document.querySelector('h1');
  return { overflow: document.documentElement.scrollWidth > document.documentElement.clientWidth, overflowBy: document.documentElement.scrollWidth - document.documentElement.clientWidth, denied: /You don't have access|Page not found/.test(document.body.innerText), crashed: /Something went wrong|Application error/.test(document.body.innerText), families, h1: h1 ? { text: h1.innerText, size: getComputedStyle(h1).fontSize, weight: getComputedStyle(h1).fontWeight } : null };
})()`;

const R = { label: LABEL, screens: {}, widths: {} };
async function main() {
  const page = (await (await fetch(`${CDP}/json/list`)).json()).find((t) => t.type === "page");
  ws = new WebSocket(page.webSocketDebuggerUrl);
  await new Promise((r) => (ws.onopen = r));
  ws.onmessage = (mm) => {
    const msg = JSON.parse(mm.data);
    if (msg.id && pending.has(msg.id)) { const p = pending.get(msg.id); pending.delete(msg.id); msg.error ? p.rej(new Error(`${p.method}: ${msg.error.message}`)) : p.res(msg.result); }
    else if (msg.method === "Fetch.requestPaused") onPaused(msg.params).catch((e) => console.error("mock", e.message));
    else if (msg.method === "Runtime.exceptionThrown") errors.push(msg.params.exceptionDetails.exception?.description?.slice(0, 200) ?? "exception");
  };
  await send("Page.enable"); await send("Runtime.enable");
  await send("Fetch.enable", { patterns: [{ urlPattern: "*/api/v1/*", requestStage: "Request" }] });

  await viewport(1440, 900);
  for (const [name, account, url] of SCREENS) {
    await as(account);
    await go(`${APP}${url}`, 2400);
    if (name === "06-permission-catalogue" && LABEL === "before") {
      const select = `document.querySelector('[data-testid=tenant-select]')`;
      if (await evaluate(`!!${select}`)) { await evaluate(`${select}.click()`); await sleep(300); }
    }
    await shot(name, true);
    R.screens[name] = await evaluate(PROBE);
  }
  if (LABEL === "after" && !process.env.QUICK) {
    for (const [w, h] of WIDTHS) {
      await viewport(w, h);
      const rows = {};
      for (const [name, account, url] of [...SCREENS, ...EXTRA.map(([a, u]) => [u.replace(`${D}/`, ""), a, u])]) {
        await as(account);
        await go(`${APP}${url}`, 1500);
        rows[name] = await evaluate(PROBE);
        if (SCREENS.some((s) => s[0] === name) && [320, 768, 1920].includes(w) && /platform-overview|fleet-overview|my-work|my-leave/.test(name)) await shot(`${name}-${w}x${h}`);
      }
      R.widths[`${w}x${h}`] = rows;
    }
  }
  if (LABEL === "after") {
    const click = async (expr) => {
      const box = await evaluate(`(() => { const e = ${expr}; if (!e) return null; e.scrollIntoView({ block: 'center' }); const r = e.getBoundingClientRect(); return { x: r.x + r.width / 2, y: r.y + r.height / 2 }; })()`);
      if (!box) return false;
      for (const type of ["mouseMoved", "mousePressed", "mouseReleased"]) await send("Input.dispatchMouseEvent", { type, x: box.x, y: box.y, button: type === "mouseMoved" ? "none" : "left", buttons: type === "mousePressed" ? 1 : 0, clickCount: type === "mouseMoved" ? 0 : 1, pointerType: "mouse" });
      await sleep(450);
      return true;
    };
    const q = (t) => `document.querySelector('[data-testid=${t}]')`;
    const option = (t) => `[...document.querySelectorAll('[role=option]')].find(o => o.innerText.includes(${JSON.stringify(t)}))`;
    const button = (t) => `[...document.querySelectorAll('button')].find(b => b.offsetParent !== null && b.textContent.trim() === ${JSON.stringify(t)})`;
    const dialogFits = `(() => { const d = document.querySelector('[role=dialog],[role=alertdialog]'); if (!d) return null; const r = d.getBoundingClientRect(); return { width: Math.round(r.width), left: Math.round(r.left), right: Math.round(r.right), viewport: innerWidth, fits: r.left >= 0 && r.right <= innerWidth, pageOverflow: document.documentElement.scrollWidth > innerWidth }; })()`;

    // Permission grid with real checkboxes, in light and in OS dark mode.
    await viewport(1440, 900);
    await as("admin");
    await go(`${APP}${D}/platform/permissions`, 2400);
    await click(q("tenant-select")); await click(option("Northline")); await sleep(900);
    await click(q("role-select")); await click(option("DRIVER")); await sleep(1400);
    const CHECK = `(() => { const boxes = [...document.querySelectorAll('[data-testid=permission-matrix] [role=checkbox]')]; const b = boxes.find(x => !x.disabled && x.getAttribute('aria-checked') === 'true') ?? boxes[0]; if (!b) return null; const s = getComputedStyle(b), r = b.getBoundingClientRect(), a = getComputedStyle(b, '::after'); return { count: boxes.length, size: [Math.round(r.width), Math.round(r.height)], clickTarget: Math.round(parseFloat(a.width)), fill: s.backgroundColor, canvas: getComputedStyle(document.body).backgroundColor, thead: getComputedStyle(document.querySelector('.dv-thead th')).backgroundColor }; })()`;
    R.checkboxLight = await evaluate(CHECK);
    await evaluate(`document.querySelector('[data-menu=driver_work]')?.scrollIntoView({ block: 'center' })`); await sleep(300);
    await shot("permission-matrix-1440x900");
    await send("Emulation.setEmulatedMedia", { features: [{ name: "prefers-color-scheme", value: "dark" }] }); await sleep(600);
    R.checkboxOsDark = await evaluate(CHECK);
    await send("Emulation.setEmulatedMedia", { features: [] });

    // Dialogs at the narrowest width.
    await viewport(320, 568);
    await as("driver");
    await go(`${APP}${D}/my-work/leave`, 1800);
    await click(button("Request leave"));
    R.dialogLeave320 = await evaluate(dialogFits);
    await shot("dialog-request-leave-320x568");
    await go(`${APP}${D}/my-work/leave`, 1800);
    await click(button("Cancel request"));
    R.dialogCancel320 = await evaluate(dialogFits);
    await shot("dialog-cancel-leave-320x568");

    await viewport(1440, 900);
    await go(`${APP}/login`, 500);
    await evaluate(`sessionStorage.clear()`);
    await go(`${APP}/login`, 1500);
    await shot("login-1440x900");
  }
  R.unmatched = [...unmatched].sort();
  R.errors = [...new Set(errors)];
  fs.writeFileSync(path.join(OUT, "results.json"), JSON.stringify({ note: "MOCKED: synthetic figures for screenshots only. Nothing reached the backend.", results: R }, null, 2));
  console.log("done");
  process.exit(0);
}
