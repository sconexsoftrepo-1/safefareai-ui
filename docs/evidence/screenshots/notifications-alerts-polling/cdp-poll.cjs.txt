async function onPaused({ requestId, request }) {
  const u = new URL(request.url);
  const p = u.pathname.replace(/^.*\/api\/v1/, "");
  const headers = Object.fromEntries(Object.entries(request.headers).map(([k, v]) => [k.toLowerCase(), v]));
  const th = headers["x-tenant-id"] ?? null;
  if (request.method === "OPTIONS") return send("Fetch.fulfillRequest", { requestId, responseCode: 204, responseHeaders: cors(), body: "" });
  const entry = { t: Date.now(), method: request.method, path: p, query: u.search, tenant: th, hasBody: !!request.postData, bearer: /^Bearer /.test(headers.authorization ?? ""), start: Date.now(), end: null };
  captured.push(entry);
  const ok = (data, total) => ({ code: 200, msg: "Success", data, total: total ?? null, extend: null });
  let status = 200, res = ok(null);
  const q = u.searchParams;
  if (p === "/auth/refresh") {
    refreshCalls++;
    if (refreshAllowed) { tokenExpired = false; res = ok({ accessToken: jwt(), tokenType: "Bearer", expiresIn: 3600, refreshToken: "synthetic-2", refreshExpiresIn: 86400 }); }
    else { status = 401; res = envelopeError(401, "INVALID_TOKEN", "Refresh token expired"); }
    entry.end = Date.now();
    return send("Fetch.fulfillRequest", { requestId, responseCode: status, responseHeaders: [...cors(), { name: "Content-Type", value: "application/json" }], body: Buffer.from(JSON.stringify(res)).toString("base64") }).catch(() => {});
  }
  if (tokenExpired && !/^\/auth\//.test(p)) {
    entry.end = Date.now();
    return send("Fetch.fulfillRequest", { requestId, responseCode: 401, responseHeaders: [...cors(), { name: "Content-Type", value: "application/json" }], body: Buffer.from(JSON.stringify(envelopeError(401, "UNAUTHENTICATED", "Token expired"))).toString("base64") }).catch(() => {});
  }
  const n = p.match(/^\/notifications\/(\d+)\/(read|unread)$/);
  const det = p.match(/^\/alerts\/(\d+)$/);
  if (delayMs && /^\/(notifications\/poll|alerts$)/.test(p)) await sleep(delayMs);
  if (p === "/me") res = ok({ id: 1, displayName: "Synthetic user", email: "synthetic@example.invalid", ...(identity.platform ? {} : { tenantName: "Ganesh Travels (synthetic)", tenantCode: "Ganesh" }) });
  else if (p === "/me/permissions") res = ok(identity);
  else if (p === "/auth/logout") res = ok(null);
  else if (netFail && /^\/(notifications\/poll|alerts$)/.test(p)) {
    entry.end = Date.now();
    if (netFail === "network") return send("Fetch.failRequest", { requestId, errorReason: "ConnectionRefused" });
    status = 503; res = envelopeError(503, "UPSTREAM_UNAVAILABLE", "Service unavailable");
  } else if (p === "/notifications/poll") res = ok(pollBody());
  else if (p === "/notifications") {
    let list = identity.platform ? [] : notes;
    if (q.get("type")) list = list.filter((x) => x.type === q.get("type"));
    if (q.get("unreadOnly") === "true") list = list.filter((x) => !x.read);
    res = ok(list, list.length);
  } else if (n) {
    const row = notes.find((x) => x.id === Number(n[1]));
    if (!row) { status = 404; res = envelopeError(404, "NOT_FOUND", "Notification not found"); }
    else { row.read = n[2] === "read"; row.readAt = row.read ? new Date().toISOString() : null; res = ok(row); }
  } else if (p === "/notifications/read-all") {
    const t = q.get("type"); let updated = 0;
    for (const x of notes) if (!x.read && (!t || x.type === t)) { x.read = true; updated++; }
    res = ok({ updated });
  } else if (p === "/alerts" && request.method === "GET") {
    if (alertsStatus) { status = alertsStatus[0]; res = envelopeError(...alertsStatus); }
    else {
      let list = alerts;
      if (q.get("kind")) list = list.filter((a) => a.kind === q.get("kind"));
      if (q.get("openOnly") === "true") list = list.filter((a) => ["NEW", "ACKNOWLEDGED", "IN_PROGRESS"].includes(a.handlingStatus));
      if (q.get("keyword")) list = list.filter((a) => a.alarmName.toLowerCase().includes(q.get("keyword").toLowerCase()));
      const size = Number(q.get("pageSize") ?? 20), page = Number(q.get("pageNumber") ?? 1);
      res = ok(list.slice((page - 1) * size, page * size), list.length);
    }
  } else if (det) {
    const a = alerts.find((x) => x.id === Number(det[1]));
    if (!a) { status = 404; res = envelopeError(404, "NOT_FOUND", "Alert not found"); }
    else res = ok({ alert: { ...a, assigneeName: "Control Room" }, actions: [{ id: 15, actionType: "ASSIGN", actorUserId: 2, actorName: "Fleet Admin", assigneeUserId: 7, assigneeName: "Control Room", note: "Please check", automatic: false, createdAt: "2026-11-04T04:25:00Z", stepNo: null }], evidence: [{ id: 44, mediaType: "IMAGE", channel: null, url: "https://media.safefare-mock.net/image.jpg", status: "PENDING", contentType: null, sizeBytes: null, durationMs: null, storedAt: null, createdAt: "2026-11-04T04:20:57Z" }] });
  } else if (/^\/alarms\//.test(p)) {
    res = { ...ok([{ id: 837, vendorAlarmId: "V837", deviceId: 1215, terminalNo: "13600000003", deviceName: "AP29Z0001", alarmTypeId: 208, alarmName: "Vendor history row", alarmTime: "2026-11-04T04:20:54Z", vendorAlarmTime: "2026-11-04 09:50:54", severity: 3, speedKmh: 42, lat: 17.42, lon: 78.44, handlingStatus: "NEW" }], 1), extend: { source: "VENDOR", stale: false, fallbackReason: null, syncedAt: null, persisted: true, message: null } };
  } else if (p === "/platform/tenants") res = ok([{ id: 1, code: "Ganesh", name: "Ganesh Travels (synthetic)", status: "ACTIVE" }], 1);
  else if (p === "/platform/switch-tenant") res = ok({ tenantId: 1, code: "Ganesh", name: "Ganesh Travels (synthetic)", header: "X-Tenant-Id" });
  else if (p === "/reference/alarm-types") res = ok([{ id: 208, label: "Pedestrian Collision", code: "PC" }], 1);
  else if (p === "/devices") res = ok([{ id: 1215, name: "AP29Z0001", terminalNo: "13600000003" }], 1);
  else res = ok([], 0);
  entry.end = Date.now();
  await send("Fetch.fulfillRequest", { requestId, responseCode: status, responseHeaders: [...cors(), { name: "Content-Type", value: "application/json" }], body: Buffer.from(JSON.stringify(res)).toString("base64") }).catch(() => {});
}

async function connect() {
  const page = (await (await fetch(`${CDP}/json/list`)).json()).find((t) => t.type === "page");
  ws = new WebSocket(page.webSocketDebuggerUrl);
  await new Promise((r) => (ws.onopen = r));
  ws.onmessage = (mm) => {
    const msg = JSON.parse(mm.data);
    if (msg.id && pending.has(msg.id)) { const p = pending.get(msg.id); pending.delete(msg.id); msg.error ? p.rej(new Error(`${p.method}: ${msg.error.message}`)) : p.res(msg.result); }
    else if (msg.method === "Fetch.requestPaused") onPaused(msg.params).catch((e) => console.error("mock", e.message));
  };
  await send("Page.enable"); await send("Runtime.enable");
  await send("Fetch.enable", { patterns: [{ urlPattern: "*/api/v1/*", requestStage: "Request" }] });
  await viewport(1366, 768);
  await go(`${APP}/login`, 1500);
  await evaluate(`sessionStorage.setItem('safefare:session:v1', ${JSON.stringify(newSession())})`);
}
