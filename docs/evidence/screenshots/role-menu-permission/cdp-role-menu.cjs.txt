async function onPaused({ requestId, request }) {
  const u = new URL(request.url);
  const p = u.pathname.replace(/^.*\/api\/v1/, "");
  const th = Object.entries(request.headers).find(([k]) => k.toLowerCase() === "x-tenant-id")?.[1] ?? null;
  const body = request.postData ? JSON.parse(request.postData) : null;
  if (request.method === "OPTIONS") return send("Fetch.fulfillRequest", { requestId, responseCode: 204, responseHeaders: cors(), body: "" });
  captured.push({ method: request.method, path: p, query: u.search, tenant: th, body, auth: /^Bearer /.test(Object.entries(request.headers).find(([k]) => k.toLowerCase() === "authorization")?.[1] ?? "") });
  const ok = (data, total) => ({ code: 200, msg: "Success", data, ...(total !== undefined ? { total } : {}) });
  let status = 200, res = ok(null);
  const m = p.match(/^\/roles\/(\d+)\/menu-permissions$/);
  const tenantId = identity.platform ? (th ? Number(th) : null) : identity.tenantId;
  if (p === "/me") res = ok({ id: 1, displayName: "Synthetic user", email: "synthetic@example.invalid", ...(identity.platform ? {} : { tenantName: "Ganesh Travels (synthetic)", tenantCode: "Ganesh" }) });
  else if (p === "/me/permissions") res = ok(identity);
  else if (p === "/platform/tenants") res = ok(TENANTS, TENANTS.length);
  else if (p === "/platform/switch-tenant") { const t = TENANTS.find((x) => x.id === body.tenantId); res = ok({ tenantId: t.id, code: t.code, name: t.name, header: "X-Tenant-Id" }); }
  else if (p === "/roles" && request.method === "GET") res = ok(ROLES.map((r) => ({ ...r, tenantId })), ROLES.length);
  else if (m) {
    const roleId = Number(m[1]);
    if (tenantId === null) { status = 400; res = envelopeError(400, "TENANT_REQUIRED", "Select a tenant (X-Tenant-Id header) for this operation"); }
    else if (request.method === "GET" && getError) { status = getError[0]; res = envelopeError(...getError); }
    else if (request.method === "GET") res = ok(grid(tenantId, roleId));
    else if (nextPutError) { status = nextPutError[0]; res = envelopeError(...nextPutError); nextPutError = null; }
    else { held[tenantId][roleId] = [...new Set(body.permissionCodes)]; res = ok(grid(tenantId, roleId)); }
  } else if (p === "/permissions" && request.method === "GET") res = ok(CATALOG, CATALOG.length);
  else if (p === "/permissions" && request.method === "POST") { CATALOG = [...CATALOG, { code: body.code, description: body.description, scope: body.scope }]; status = 201; res = { code: 201, msg: "Created", data: body }; }
  else if (/^\/permissions\//.test(p) && request.method === "PUT") { const code = decodeURIComponent(p.split("/")[2]); CATALOG = CATALOG.map((c) => (c.code === code ? { code, ...body } : c)); res = ok({ code, ...body }); }
  else if (/^\/permissions\//.test(p) && request.method === "DELETE") { const code = decodeURIComponent(p.split("/")[2]); CATALOG = CATALOG.filter((c) => c.code !== code); res = ok(null); }
  await send("Fetch.fulfillRequest", { requestId, responseCode: status, responseHeaders: [...cors(), { name: "Content-Type", value: "application/json" }], body: Buffer.from(JSON.stringify(res)).toString("base64") });
}
