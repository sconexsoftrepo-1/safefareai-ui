async function onPaused({ requestId, request }) {
  const u = new URL(request.url);
  const p = u.pathname.replace(/^.*\/api\/v1/, "");
  const th = Object.entries(request.headers).find(([k]) => k.toLowerCase() === "x-tenant-id")?.[1] ?? null;
  captured.push({ method: request.method, path: p, query: u.search, tenant: th, body: request.postData ?? null });
  if (request.method === "OPTIONS") return send("Fetch.fulfillRequest", { requestId, responseCode: 204, responseHeaders: cors(), body: "" });
  let status = 200, body;
  const ok = (data, total) => ({ code: 200, msg: "Success", data, ...(total !== undefined ? { total } : {}) });
  if (p === "/auth/login") body = ok({ accessToken: jwt(), refreshToken: jwt(), tokenType: "Bearer", expiresIn: 3600 });
  else if (p === "/me") body = ok({ id: identity.userId, displayName: "Synthetic user", email: "synthetic@example.invalid" });
  else if (p === "/me/permissions") body = ok(identity);
  else if (p === "/platform/tenants" && request.method === "GET") body = ok(TENANTS, TENANTS.length);
  else if (p === "/platform/tenants" && request.method === "POST") {
    const req = JSON.parse(request.postData);
    if (createMode === "slow") await sleep(2500);
    if (createMode === "duplicate") { status = 409; body = { code: 409, msg: "Tenant code already exists (MOCKED)", data: null, total: null, extend: { errorCode: "DUPLICATE", requestId: "mock-409" } }; }
    else if (createMode === "validation") { status = 400; body = { code: 400, msg: "Request validation failed (MOCKED)", data: null, total: null, extend: { errorCode: "VALIDATION_FAILED", fieldErrors: [{ field: "s3Prefix", message: "must match the storage prefix format (MOCKED)" }], requestId: "mock-400" } }; }
    else {
      const t = { id: 9001 + TENANTS.length, code: req.code || req.name.replace(/[^A-Za-z]/g, "").slice(0, 6).toUpperCase(), name: req.name, status: req.status, timezone: req.timezone, s3Prefix: req.s3Prefix, users: 0, devices: 0, drivers: 0, current: false, createdAt: "2026-11-02T11:45:15.172Z", updatedAt: "2026-11-02T11:45:15.172Z" };
      TENANTS = [...TENANTS, t];
      status = 201; body = { code: 201, msg: "Success", data: t, total: null, extend: null };

}
