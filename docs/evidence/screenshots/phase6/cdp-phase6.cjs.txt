async function onPaused({ requestId, request }) {
  const u = new URL(request.url);
  const p = u.pathname.replace(/^.*\/api\/v1/, "");
  const th = Object.entries(request.headers).find(([k]) => k.toLowerCase() === "x-tenant-id")?.[1] ?? null;
  captured.push({ method: request.method, path: p, query: u.search, tenant: th, body: request.postData ?? null });
  if (request.method === "OPTIONS") return send("Fetch.fulfillRequest", { requestId, responseCode: 204, responseHeaders: cors(), body: "" });
  let status = 200, body;
  const ok = (data, total) => ({ code: 200, msg: "Success", data, ...(total !== undefined ? { total } : {}) });
  if (p === "/auth/login") body = ok({ accessToken: jwt(), refreshToken: jwt(), tokenType: "Bearer", expiresIn: 3600 });
  else if (p === "/me") body = ok({ id: identity.userId, displayName: "Synthetic user", email: "synthetic@example.invalid" });
  else if (p === "/me/permissions") body = ok(identity);
  else if (p === "/platform/tenants") {
    if (tenantsMode === "error") { status = 503; body = { code: 503, msg: "Database unavailable (MOCKED)", data: null, total: null, extend: { errorCode: "SERVICE_UNAVAILABLE", requestId: "mock-503" } }; }
    else if (tenantsMode === "empty") body = ok([], 0);
    else body = ok(TENANTS, TENANTS.length);
  } else if (p === "/reference/statuses") body = ok(CATALOG);
  else if (p === "/platform/switch-tenant") { const id = JSON.parse(request.postData).tenantId; const t = TENANTS.find((x) => x.id === id); body = ok({ tenantId: id, code: t.code, name: t.name, header: "X-Tenant-Id" }); }
  else if (p === "/drivers") { const d = DRIVERS[th ?? "all"] ?? []; body = ok(d, d.length); }
  else body = ok(null);
  await send("Fetch.fulfillRequest", { requestId, responseCode: status, responseHeaders: [...cors(), { name: "Content-Type", value: "application/json" }], body: Buffer.from(JSON.stringify(body)).toString("base64") });
}
async function viewport(w, h) { await send("Emulation.setDeviceMetricsOverride", { width: w, height: h, deviceScaleFactor: 1, mobile: w < 768 }); await send("Emulation.setTouchEmulationEnabled", { enabled: w < 768 }); }
async function go(url, ms = 2500) { await send("Page.navigate", { url }); await sleep(ms); }
async function shot(name, full = false) {
  let extra = {};
  if (full) { const mm = await send("Page.getLayoutMetrics"); extra = { clip: { x: 0, y: 0, width: mm.cssContentSize.width, height: Math.min(mm.cssContentSize.height, 5000), scale: 1 }, captureBeyondViewport: true }; }
  const r = await send("Page.captureScreenshot", { format: "png", ...extra });
  fs.writeFileSync(path.join(OUT, `${name}.png`), Buffer.from(r.data, "base64"));
}
async function typeInto(sel, text) { await evaluate(`(() => { const el = document.querySelector(${JSON.stringify(sel)}); el.focus(); el.select(); return true; })()`); if (text) await send("Input.insertText", { text }); else { await send("Input.dispatchKeyEvent", { type: "keyDown", key: "Backspace", code: "Backspace", windowsVirtualKeyCode: 8 }); await send("Input.dispatchKeyEvent", { type: "keyUp", key: "Backspace", code: "Backspace", windowsVirtualKeyCode: 8 }); } }
const click = (sel) => evaluate(`(() => { const el = document.querySelector(${JSON.stringify(sel)}); if (!el) return false; el.click(); return true; })()`);
const clickText = (t, scope = "button") => evaluate(`(() => { const el = [...document.querySelectorAll(${JSON.stringify(scope)})].find(b => b.offsetParent !== null && b.textContent.trim().startsWith(${JSON.stringify(t)})); if (!el) return false; el.click(); return true; })()`);
const clickLabel = (l) => evaluate(`(() => { const el = [...document.querySelectorAll('[aria-label]')].find(b => b.offsetParent !== null && b.getAttribute('aria-label') === ${JSON.stringify(l)}); if (!el) return false; el.click(); return true; })()`);

const R = {};
async function main() {
  const page = (await (await fetch(`${CDP}/json/list`)).json()).find((t) => t.type === "page");
  ws = new WebSocket(page.webSocketDebuggerUrl);
  await new Promise((r) => (ws.onopen = r));
  ws.onmessage = (mm) => { const msg = JSON.parse(mm.data); if (msg.id && pending.has(msg.id)) { const p = pending.get(msg.id); pending.delete(msg.id); msg.error ? p.rej(new Error(`${p.method}: ${msg.error.message}`)) : p.res(msg.result); } else if (msg.method === "Fetch.requestPaused") onPaused(msg.params).catch((e) => console.error("mock", e.message)); };
  await send("Page.enable"); await send("Runtime.enable");
  await send("Fetch.enable", { patterns: [{ urlPattern: "*/api/v1/*", requestStage: "Request" }] });
  await go(`${APP}/login`, 1500);

  // 1) List rendering, summary, filters, missing data, paging.
  identity = IDS.sfa;
  await viewport(1366, 768);
  await login("platform");
  let i = captured.length;
  await go(`${APP}/drive-vision/tenants`, 3000);
  const s0 = await evaluate(STATE);
  const r0 = await rows();
  R.list = { summary: s0.summary, filters: s0.filters, range: s0.range, rowsOnPage1: r0.length, firstRows: r0.slice(0, 4), missingRow: r0.find((r) => r.name === "Missing Fields Co"), unsupportedNote: s0.unsupported, tenantListHeaders: since(i).filter((c) => c.path === "/platform/tenants").map((c) => c.tenant), calls: [...new Set(since(i).map((c) => `${c.method} ${c.path}`))] };
  await shot("tenants-list-1366x768", true);
  await clickLabel("Next page"); await sleep(400);
  const s1 = await evaluate(STATE);
  R.paging = { page2Range: s1.range, page2Rows: (await rows()).length };

  // Search and status filter (including a documented-but-unused catalogue status).
  await typeInto("input[aria-label='Search tenants']", "blue"); await sleep(300);
  R.search = { blue: (await rows()).map((r) => r.name), range: (await evaluate(STATE)).range };
  await typeInto("input[aria-label='Search tenants']", "miss"); await sleep(300);
  R.search.miss = (await rows()).map((r) => r.name);
  await typeInto("input[aria-label='Search tenants']", ""); await sleep(300);
  await clickText("INACTIVE", "[aria-label='Filter by status'] button"); await sleep(300);
  R.filter = { inactive: (await rows()).map((r) => `${r.name}:${r.action}`) };
  await clickText("ARCHIVED", "[aria-label='Filter by status'] button"); await sleep(300);
  R.filter.archived = { rows: (await rows()).length, empty: (await evaluate(STATE)).text.includes("No matching tenants") };
  await clickText("No status", "[aria-label='Filter by status'] button"); await sleep(300);
  R.filter.noStatus = (await rows()).map((r) => r.name);
  await clickText("All", "[aria-label='Filter by status'] button"); await sleep(300);

  // 2) Details drawer (from list data) — viewing doesn't select; select from drawer → switcher follows.
  await clickLabel("Details for Missing Fields Co"); await sleep(800);
  const d0 = await evaluate(STATE);
  R.details = { missingFields: d0.detail, switcherBefore: d0.switcher };
  await shot("tenant-detail-missing-1366x768");
  await send("Input.dispatchKeyEvent", { type: "keyDown", key: "Escape", code: "Escape", windowsVirtualKeyCode: 27 }); await sleep(500);
  await clickLabel("Details for Acme Logistics"); await sleep(800);
  i = captured.length;
  await evaluate(`(() => { const b = [...document.querySelectorAll('[role=dialog] button')].find(x => x.getAttribute('aria-label') === 'Select Acme Logistics'); b && b.click(); return !!b; })()`);
  await sleep(2500);
  const d1 = await evaluate(STATE);
  R.selectFromPage = { switcherAfter: d1.switcher, context: d1.context, switchCalls: since(i).filter((c) => c.path === "/platform/switch-tenant").map((c) => ({ body: c.body, tenantHeader: c.tenant })), detailNowSays: (d1.detail || "").includes("Selected in this tab") };
  await send("Input.dispatchKeyEvent", { type: "keyDown", key: "Escape", code: "Escape", windowsVirtualKeyCode: 27 }); await sleep(500);

  // 3) Switch via top-bar → page reflects it (shared state); then tenant data on Drivers (client-side nav), no stale data.
  await click("[data-testid=tenant-switcher]"); await sleep(1200);
  await clickText("Blue Line Cabs", "[data-radix-popper-content-wrapper] li button"); await sleep(2500);
  const r1 = await rows();
  R.sharedState = { pageSelectedRow: r1.filter((r) => r.action === "Selected").map((r) => r.name), context: (await evaluate(STATE)).context };
  i = captured.length;
  await click("a[href='/drive-vision/drivers']"); await sleep(2500);
  const dv = await evaluate(STATE);
  R.noStale = { driversHeader: since(i).filter((c) => c.path === "/drivers").map((c) => c.tenant), showsBlue: dv.text.includes("Blue Driver"), showsAcme: dv.text.includes("Acme Driver"), showsAll: dv.text.includes("All-tenant Driver") };
  await click("a[href='/drive-vision/tenants']"); await sleep(2500);
  await clickText("Return to all tenants"); await sleep(1200);
  R.returnToAll = { context: (await evaluate(STATE)).context, switcher: (await evaluate(STATE)).switcher };

  // 4) Errors: initial failure → error panel (not empty); refresh failure with data → stale notice; empty list.
  tenantsMode = "error";
  await go(`${APP}/drive-vision/tenants`, 3000);
  const e0 = await evaluate(STATE);
  R.errorInitial = { alerts: e0.alerts, emptyStateShown: e0.text.includes("No tenants returned") };

  await go(`${APP}/drive-vision/tenants`, 3000);
  R.empty = { text: (await evaluate(STATE)).text.includes("No tenants returned") };

  R.responsive = {};

    const ld = await evaluate(LAYOUT);
    await shot(`tenant-detail-${w}x${h}`);
    await send("Input.dispatchKeyEvent", { type: "keyDown", key: "Escape", code: "Escape", windowsVirtualKeyCode: 27 }); await sleep(500);
    await click("[data-testid=tenant-switcher]"); await sleep(1000);
    const ls = await evaluate(LAYOUT);
    const pop = await evaluate(`(() => { const p = document.querySelector('[data-radix-popper-content-wrapper]'); if (!p) return null; const b = p.getBoundingClientRect(); return { inView: b.left >= 0 && b.right <= document.documentElement.clientWidth }; })()`);

  }

  R.anyWrite = captured.filter((c) => !["GET", "OPTIONS"].includes(c.method) && !["/auth/login", "/platform/switch-tenant"].includes(c.path)).map((c) => `${c.method} ${c.path}`);
  fs.writeFileSync(path.join(OUT, "results.json"), JSON.stringify(R, null, 2));
  console.log(JSON.stringify(R, null, 1));
  ws.close();
}
main().catch((e) => { console.error(e); process.exit(1); });
