async function onPaused({ requestId, request }) {
  const u = new URL(request.url);
  const p = u.pathname.replace(/^.*\/api\/v1/, "");
  const th = Object.entries(request.headers).find(([k]) => k.toLowerCase() === "x-tenant-id")?.[1] ?? null;
  const body = request.postData ? JSON.parse(request.postData) : null;
  captured.push({ method: request.method, path: p, query: u.search, tenant: th, body });
  if (request.method === "OPTIONS") return send("Fetch.fulfillRequest", { requestId, responseCode: 204, responseHeaders: cors(), body: "" });
  const ok = (data, total) => ({ code: 200, msg: "Success", data, ...(total !== undefined ? { total } : {}) });
  const err = (status, errorCode, msg, fieldErrors) => [status, { code: status, msg, data: null, extend: { errorCode, requestId: `mock-${status}`, ...(fieldErrors ? { fieldErrors } : {}) } }];
  let status = 200, res;
  const dm = p.match(/^\/drivers\/(\d+)$/), am = p.match(/^\/drivers\/(\d+)\/assignments$/), em = p.match(/^\/drivers\/(\d+)\/assignments\/(\d+)\/end$/), vm = p.match(/^\/devices\/(\d+)\/driver-assignments$/);
  if (p === "/auth/login") res = ok({ accessToken: jwt(), refreshToken: jwt(), tokenType: "Bearer", expiresIn: 3600 });
  else if (p === "/me") res = ok({ id: 1, displayName: "Synthetic user", email: "synthetic@example.invalid" });
  else if (p === "/me/permissions") res = ok(identity);
  else if (p === "/platform/tenants") res = ok(TENANTS, 2);
  else if (p === "/platform/switch-tenant") { const t = TENANTS.find((x) => x.id === body.tenantId); res = ok({ tenantId: t.id, code: t.code, name: t.name, header: "X-Tenant-Id" }); }
  else if (p === "/drivers" && request.method === "GET") {
    if (mode.drivers === "error") [status, res] = err(503, "SERVICE_UNAVAILABLE", "Driver service unavailable (MOCKED)");
    else if (mode.drivers === "empty") res = ok([], 0);
    else { const page = Number(u.searchParams.get("pageNumber") || 1); let rows = driversFor(th ?? "all", page); const kw = u.searchParams.get("keyword"); if (kw) rows = rows.filter((r) => r.name.toLowerCase().includes(kw.toLowerCase())); const st = u.searchParams.get("status"); if (st) rows = rows.filter((r) => r.status === st); res = ok(rows, kw || st ? rows.length : 25); }
  } else if (p === "/drivers" && request.method === "POST") {
    if (mode.create === "invalid") [status, res] = err(400, "VALIDATION_FAILED", "Request validation failed (MOCKED)", [{ field: "email", message: "Email already used (MOCKED)" }]);
    else res = ok({ id: 99, name: body.name, status: body.status });
  } else if (dm && request.method === "GET") res = ok({ ...driversFor(th ?? "all", 1)[0], id: Number(dm[1]), name: Number(dm[1]) === 1 ? "Acme Driver 01" : `Driver ${dm[1]}`, currentAssignment: (assignments[dm[1]] ?? []).find((a) => a.active), hiredOn: "2024-04-01", userId: undefined });
  else if (am && request.method === "GET") res = ok(assignments[am[1]] ?? [], (assignments[am[1]] ?? []).length);
  else if (am && request.method === "POST") {
    if (mode.assign === "conflict") [status, res] = err(409, "OPERATION_NOT_ALLOWED", "Vehicle already assigned (MOCKED)");
    else { const list = (assignments[am[1]] ??= []); if (body.replaceCurrent) list.forEach((a) => { if (a.active) { a.active = false; a.assignedTo = new Date().toISOString(); } }); const dev = DEVICES.find((d) => d.id === body.deviceId); const a = { id: 30 + list.length, deviceId: body.deviceId, deviceName: dev?.name, driverId: Number(am[1]), driverName: "Acme Driver 01", assignedFrom: body.from ?? new Date().toISOString(), assignedTo: body.to, active: true, source: "MANUAL" }; list.unshift(a); res = ok(a); }
  } else if (em) { const a = (assignments[em[1]] ?? []).find((x) => x.id === Number(em[2])); if (a) { a.active = false; a.assignedTo = body?.at ?? new Date().toISOString(); } res = ok(a); }
  else if (vm) res = ok((deviceAssignments[vm[1]] ?? (() => []))(), 1);
  else if (p === "/devices") res = ok(DEVICES, DEVICES.length);
  else if (/\/safety-score$/.test(p)) res = ok({ score: { score: 82, grade: "B", alarmCount: 4, severeCount: 1, distanceKm: 512 } });
  else if (/\/safety-events$/.test(p)) res = ok([{ id: 1, alarmName: "Harsh braking (synthetic)", alarmTime: ago(3) }], 1);
  else res = ok(null);
  await send("Fetch.fulfillRequest", { requestId, responseCode: status, responseHeaders: [...cors(), { name: "Content-Type", value: "application/json" }], body: Buffer.from(JSON.stringify(res)).toString("base64") });
}
