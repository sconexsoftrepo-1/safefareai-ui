async function onPaused({ requestId, request }) {
  const u = new URL(request.url);
  const p = u.pathname.replace(/^.*\/api\/v1/, "");
  captured.push({ method: request.method, path: p, query: u.search, tenantHeader: Object.keys(request.headers).some((h) => h.toLowerCase() === "x-tenant-id"), auth: !!request.headers.Authorization });
  if (request.method === "OPTIONS") return send("Fetch.fulfillRequest", { requestId, responseCode: 204, responseHeaders: corsHeaders(), body: "" });
  let status = 200, body;
  if (p === "/auth/login") body = { code: 200, msg: "Success", data: { accessToken: jwt(), refreshToken: jwt(), tokenType: "Bearer", expiresIn: 3600 } };
  else if (p === "/me") body = { code: 200, msg: "Success", data: { id: scenario.identity.userId, displayName: `Synthetic ${scenario.identity.roleCode}`, email: "synthetic@example.invalid" } };
  else if (p === "/me/permissions") body = { code: 200, msg: "Success", data: scenario.identity };
  else if (scenario.overrides[p]) [status, body] = scenario.overrides[p];
  else if (DATA[p]) body = { code: 200, msg: "Success", ...DATA[p]() };
  else body = { code: 200, msg: "Success", data: null };
  await send("Fetch.fulfillRequest", { requestId, responseCode: status, responseHeaders: [...corsHeaders(), { name: "Content-Type", value: "application/json" }], body: Buffer.from(JSON.stringify(body)).toString("base64") });
}

async function viewport(w, h) { await send("Emulation.setDeviceMetricsOverride", { width: w, height: h, deviceScaleFactor: 1, mobile: w < 768 }); await send("Emulation.setTouchEmulationEnabled", { enabled: w < 768 }); }
async function go(url, ms = 2500) { await send("Page.navigate", { url }); await sleep(ms); }
async function shot(name, full = false) {
  let clip;
  if (full) { const m = await send("Page.getLayoutMetrics"); clip = { x: 0, y: 0, width: m.cssContentSize.width, height: Math.min(m.cssContentSize.height, 6000), scale: 1 }; }
  const r = await send("Page.captureScreenshot", { format: "png", ...(full ? { clip, captureBeyondViewport: true } : {}) });
  fs.writeFileSync(path.join(OUT, `${name}.png`), Buffer.from(r.data, "base64"));
}
async function typeInto(sel, text) { await evaluate(`(() => { const el = document.querySelector(${JSON.stringify(sel)}); el.focus(); el.select(); return true; })()`); await send("Input.insertText", { text }); }
const clickText = (t) => evaluate(`(() => { const el = [...document.querySelectorAll("button")].find(b => b.textContent.trim() === ${JSON.stringify(t)}); if (!el) return false; el.click(); return true; })()`);
async function login(mode) {
  await evaluate("sessionStorage.clear(), true");
  await go(`${APP}/login`, 1500);
  if (mode === "platform") await clickText("Safe-Fare Admin");
  else { await typeInto("input[name=tenantCode]", "SYNTH"); }
  await typeInto("input[name=login]", "synthetic@example.invalid");
  await typeInto("input[name=password]", "not-a-real-password");
  await clickText("Sign in");
  await sleep(4000);
}

async function main() {

  ws = new WebSocket(page.webSocketDebuggerUrl);
  await new Promise((r) => (ws.onopen = r));
  ws.onmessage = (m) => { const msg = JSON.parse(m.data); if (msg.id && pending.has(msg.id)) { const p = pending.get(msg.id); pending.delete(msg.id); msg.error ? p.rej(new Error(`${p.method}: ${msg.error.message}`)) : p.res(msg.result); } else if (msg.method === "Fetch.requestPaused") onPaused(msg.params).catch((e) => console.error("mock", e.message)); };
  await send("Page.enable"); await send("Runtime.enable");
  await send("Fetch.enable", { patterns: [{ urlPattern: "*/api/v1/*", requestStage: "Request" }] });
  await go(`${APP}/login`, 1500);

  // 1) Synthetic Safe-Fare Admin: landing + all viewports.
  scenario = { identity: IDS.safeFareAdmin, overrides: {} };
  await viewport(1440, 900);
  const start = captured.length;
  await login("platform");
  const landed = await evaluate(PROBE);
  const sfa = { landing: landed.path, scope: landed.scope, navLinks: landed.navLinks, viewports: {} };
  for (const [w, h] of VIEWPORTS) {
    await viewport(w, h);
    await go(`${APP}/drive-vision/platform`, 3000);
    const pr = await evaluate(PROBE);
    sfa.viewports[`${w}x${h}`] = { overflow: pr.horizontalOverflow, scrollWidth: pr.scrollWidth, clipped: pr.clipped, overlaps: pr.overlaps, smallTargets: pr.smallTargets, alerts: pr.alerts };
    await shot(`platform-${w}x${h}`, true);
  }
  sfa.text = (await evaluate(PROBE)).text;
  await viewport(1440, 900);
  await go(`${APP}/drive-vision/overview`, 3500);
  const ov = await evaluate(PROBE);
  sfa.overviewScopeBadge = ov.overviewScope;
  await shot("sfa-overview-1440x900");
  sfa.requests = captured.slice(start).filter((c) => c.method !== "OPTIONS");
  sfa.tenantSelectorSent = sfa.requests.filter((c) => c.tenantHeader || /tenantId=/.test(c.query)).map((c) => c.path);
  results.scenarios.safeFareAdmin = { ...sfa, requests: [...new Set(sfa.requests.map((c) => `${c.method} ${c.path}`))] };

  // 2) Partial failure: safety 503, groups 403, trends network-ish 500 — others still render.

  for (const [w, h] of [[1440, 900], [375, 812]]) {
    await viewport(w, h);
    await go(`${APP}/drive-vision/platform`, 3500);
    const pr = await evaluate(PROBE);
    results.scenarios[`partialFailure-${w}x${h}`] = { overflow: pr.horizontalOverflow, alerts: pr.alerts, stillShown: { tenants: /Acme Logistics/.test(pr.text), vehicles: /155/.test(pr.text), trend: /Daily trend/.test(pr.text) } };
    await shot(`platform-partial-failure-${w}x${h}`, true);
  }

  // 3) Empty platform: no tenants, empty dashboards.
  scenario = { identity: IDS.safeFareAdmin, overrides: { "/platform/tenants": [200, { code: 200, msg: "Success", data: [], total: 0 }], "/dashboard/overview": [200, { code: 200, msg: "Success", data: {} }], "/dashboard/fleet-status": [200, { code: 200, msg: "Success", data: [] }], "/dashboard/trends": [200, { code: 200, msg: "Success", data: [] }], "/dashboard/safety-events": [200, { code: 200, msg: "Success", data: {} }], "/groups": [200, { code: 200, msg: "Success", data: [], total: 0 }] } };
