async function onPaused({ requestId, request }) {
  const u = new URL(request.url);
  const p = u.pathname.replace(/^.*\/api\/v1/, "");
  const th = Object.entries(request.headers).find(([k]) => k.toLowerCase() === "x-tenant-id")?.[1] ?? null;
  captured.push({ method: request.method, path: p, query: u.search, tenant: th, body: request.postData ? JSON.parse(request.postData) : null });
  if (request.method === "OPTIONS") return send("Fetch.fulfillRequest", { requestId, responseCode: 204, responseHeaders: corsHeaders(), body: "" });
  const scope = th ?? "all";
  let status = 200, data = null, total;
  if (p === "/auth/login") data = { accessToken: jwt(), refreshToken: jwt(), tokenType: "Bearer", expiresIn: 3600 };
  else if (p === "/me") data = { id: identity.userId, displayName: `Synthetic ${identity.platform ? "Safe-Fare Admin" : "Tenant Admin"}`, email: "synthetic@example.invalid" };
  else if (p === "/me/permissions") data = identity;
  else if (p === "/platform/tenants") { data = TENANTS; total = 3; }
  else if (p === "/platform/switch-tenant") {
    const id = JSON.parse(request.postData).tenantId;
    if (switchOverrides[id]) [status, data] = switchOverrides[id];
    else { const t = TENANTS.find((x) => x.id === id); data = { tenantId: t.id, code: t.code, name: t.name, header: "X-Tenant-Id" }; }
  } else if (p === "/dashboard/overview") data = { vehicles: { total: VEH[scope], online: Math.round(VEH[scope] * 0.6), offline: VEH[scope] - Math.round(VEH[scope] * 0.6), driving: 10, idling: 5, parked: 20 }, trips: { tripsToday: VEH[scope] }, health: {}, operations: {} };
  else if (p === "/drivers") { data = DRIVERS[scope] ?? []; total = data.length; }
  else if (p === "/groups") { data = []; total = 0; }
  const body = status >= 400 ? data : { code: 200, msg: "Success", data, ...(total !== undefined ? { total } : {}) };
  await send("Fetch.fulfillRequest", { requestId, responseCode: status, responseHeaders: [...corsHeaders(), { name: "Content-Type", value: "application/json" }], body: Buffer.from(JSON.stringify(body)).toString("base64") });
}

async function viewport(w, h) { await send("Emulation.setDeviceMetricsOverride", { width: w, height: h, deviceScaleFactor: 1, mobile: w < 768 }); await send("Emulation.setTouchEmulationEnabled", { enabled: w < 768 }); }
async function go(url, ms = 2500) { await send("Page.navigate", { url }); await sleep(ms); }
async function shot(name, full = false) {
  let extra = {};
  if (full) { const m = await send("Page.getLayoutMetrics"); extra = { clip: { x: 0, y: 0, width: m.cssContentSize.width, height: Math.min(m.cssContentSize.height, 5000), scale: 1 }, captureBeyondViewport: true }; }
  const r = await send("Page.captureScreenshot", { format: "png", ...extra });
  fs.writeFileSync(path.join(OUT, `${name}.png`), Buffer.from(r.data, "base64"));
}
async function typeInto(sel, text) { await evaluate(`(() => { const el = document.querySelector(${JSON.stringify(sel)}); el.focus(); el.select(); return true; })()`); await send("Input.insertText", { text }); }
const clickSel = (sel) => evaluate(`(() => { const el = document.querySelector(${JSON.stringify(sel)}); if (!el) return false; el.click(); return true; })()`);
const clickText = (t, scope = "button") => evaluate(`(() => { const el = [...document.querySelectorAll(${JSON.stringify(scope)})].find(b => b.textContent.trim().startsWith(${JSON.stringify(t)})); if (!el) return false; el.click(); return true; })()`);
