async function onPaused({ requestId, request }) {
  const u = new URL(request.url);
  if (u.host === "media.gpstracktech.com") {
    mediaRequests.push(u.pathname + u.search);
    const name = u.pathname.slice(1);
    if (name === "flaky.png" && flakyDown) return send("Fetch.failRequest", { requestId, errorReason: "ConnectionFailed" });
    const allow = !["nocors.png", "hidden.png"].includes(name);
    const headers = (type) => [...(allow ? cors() : []), { name: "Content-Type", value: type }];
    const reply = (status, type, body) => send("Fetch.fulfillRequest", { requestId, responseCode: status, responseHeaders: headers(type), body: body.toString("base64") });
    if (request.method === "OPTIONS") return send("Fetch.fulfillRequest", { requestId, responseCode: allow ? 204 : 403, responseHeaders: allow ? cors() : [], body: "" });
    if (name === "ok.png" || name === "nocors.png" || name === "flaky.png") return reply(200, "image/png", media.png);
    if (name === "ok.webm") return reply(200, "video/webm", media.webm);
    if (name === "corrupt.png") return reply(200, "image/png", Buffer.from("not an image"));
    if (name === "expired.webm") return reply(403, "text/plain", Buffer.from("Expired"));
    return reply(404, "text/plain", Buffer.from("Not found"));
  }
  const p = u.pathname.replace(/^.*\/api\/v1/, "");
  if (request.method === "OPTIONS") return send("Fetch.fulfillRequest", { requestId, responseCode: 204, responseHeaders: [...cors(), { name: "Access-Control-Allow-Headers", value: "Authorization, Content-Type, X-Request-Id, X-Tenant-Id" }], body: "" });
  const ok = (data, total) => ({ code: 200, msg: "Success", data, ...(total !== undefined ? { total } : {}) });
  let res = ok([], 0);
  if (p === "/me") res = ok({ id: 2, displayName: "Synthetic admin", email: "admin@example.invalid" });
  else if (p === "/me/permissions") res = ok({ ...IDENTITIES.tenantAdmin, userId: 2, tenantId: 1, homeTenantId: 1 });
  else if (p === "/alerts") res = ok([ALERT], 1);
  else if (p === "/alerts/501") res = ok({ alert: ALERT, actions: [], evidence: EVIDENCE });
  else if (p === "/alarms/ai") res = ok([ALARM], 1);
  await send("Fetch.fulfillRequest", { requestId, responseCode: 200, responseHeaders: [...cors(), { name: "Content-Type", value: "application/json" }], body: Buffer.from(JSON.stringify(res)).toString("base64") });
}

async function mouseClick(expr, wait = 450) {
  const box = await evaluate(`(() => { const e = ${expr}; if (!e) return null; e.scrollIntoView({ block: 'center' }); const r = e.getBoundingClientRect(); return { x: r.x + r.width / 2, y: r.y + r.height / 2 }; })()`);
  if (!box) throw new Error(`no element: ${expr}`);
  for (const type of ["mouseMoved", "mousePressed", "mouseReleased"]) await send("Input.dispatchMouseEvent", { type, x: box.x, y: box.y, button: type === "mouseMoved" ? "none" : "left", buttons: type === "mousePressed" ? 1 : 0, clickCount: type === "mouseMoved" ? 0 : 1, pointerType: "mouse" });
  await sleep(wait);
  return true;
}
